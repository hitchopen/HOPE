cmake_minimum_required(VERSION 3.16)
project(hope_field_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)

add_executable(hope-runner-transport-relay-native runner_transport.cpp)
ament_target_dependencies(hope-runner-transport-relay-native rclcpp std_msgs)
add_executable(hope-imu-telemetry imu_telemetry.cpp)
ament_target_dependencies(hope-imu-telemetry rclcpp std_msgs sensor_msgs)
foreach(target hope-runner-transport-relay-native hope-imu-telemetry)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${target} RUNTIME DESTINATION libexec)
endforeach()
