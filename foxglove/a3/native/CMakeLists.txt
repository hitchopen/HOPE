cmake_minimum_required(VERSION 3.16)
project(hope_field_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_TESTING "Build maintained isolated regression tools (never installed)" OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
option(HOPE_MONITOR_REQUIRE_VENDOR_RPC "Require A3 vendor emergency RPC at build time" ON)
find_package(ros2_plugin_proto QUIET)
if(HOPE_MONITOR_REQUIRE_VENDOR_RPC AND NOT ros2_plugin_proto_FOUND)
  message(FATAL_ERROR "Source the A3 vendor ROS prefix: ros2_plugin_proto is required for production")
endif()

add_executable(hope-transport-gateway transport_gateway.cpp)
ament_target_dependencies(hope-transport-gateway rclcpp std_msgs)
add_executable(hope-base-pose-transport-relay base_pose_transport.cpp)
ament_target_dependencies(hope-base-pose-transport-relay rclcpp std_msgs)
add_executable(hope-imu-telemetry imu_telemetry.cpp)
ament_target_dependencies(hope-imu-telemetry rclcpp std_msgs sensor_msgs)
add_library(hope-mocap-config-transport-core STATIC mocap_config_transport.cpp)
ament_target_dependencies(hope-mocap-config-transport-core rclcpp std_msgs)
target_compile_options(hope-mocap-config-transport-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hope-mocap-config-transport mocap_config_main.cpp)
target_link_libraries(hope-mocap-config-transport hope-mocap-config-transport-core)
add_executable(hope-monitor monitor.cpp)
ament_target_dependencies(hope-monitor rclcpp std_msgs sensor_msgs geometry_msgs std_srvs tf2_ros tf2_msgs visualization_msgs)
if(ros2_plugin_proto_FOUND)
  ament_target_dependencies(hope-monitor ros2_plugin_proto)
  target_compile_definitions(hope-monitor PRIVATE HOPE_MONITOR_VENDOR_RPC=1)
else()
  message(WARNING "hope-monitor: vendor RPC not compiled; Runner-only degraded path. Do not use this build to replace a full vendor-capable monitor.")
endif()
foreach(target hope-transport-gateway hope-base-pose-transport-relay hope-imu-telemetry hope-mocap-config-transport hope-monitor)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${target} RUNTIME DESTINATION libexec)
endforeach()
include(CTest)
if(BUILD_TESTING)
  # Deployment can copy a3/native alone. Tests explicitly require a full checkout;
  # no test drivers or fault injection sources belong in the deployment folder.
  set(native_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests")
  foreach(test_source native/monitor_core_test.cpp fixtures/monitor_network_driver.cpp
      fixtures/monitor_probe_faults.cpp fixtures/mocap_config_transport_driver.cpp)
    if(NOT EXISTS "${native_tests}/${test_source}")
      message(FATAL_ERROR "BUILD_TESTING=ON requires foxglove/tests from the same checkout: missing ${test_source}")
    endif()
  endforeach()
  add_executable(hope-monitor-core-test "${native_tests}/native/monitor_core_test.cpp")
  target_include_directories(hope-monitor-core-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(hope-monitor-core-test PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(hope-monitor-core-test pthread)
  add_test(NAME hope-monitor-core COMMAND hope-monitor-core-test)
  add_executable(hope-monitor-network-test-driver "${native_tests}/fixtures/monitor_network_driver.cpp")
  target_include_directories(hope-monitor-network-test-driver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  # Fault injection is a test-only shared object, never a production dependency
  # and deliberately not installed with the executables.
  add_library(hope-monitor-probe-faults SHARED "${native_tests}/fixtures/monitor_probe_faults.cpp")
  target_link_libraries(hope-monitor-probe-faults dl)
  add_executable(hope-mocap-config-transport-test-driver "${native_tests}/fixtures/mocap_config_transport_driver.cpp")
  target_include_directories(hope-mocap-config-transport-test-driver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(hope-mocap-config-transport-test-driver hope-mocap-config-transport-core)
endif()
